clean up sensor data

sensor_data <- sensory_raw|>
  mutate(
    #parse the messy charcter string to correct date - (day/month/year hour:minute)
    datetime = dmy_hm(timestamp),
    
    #convert the -999.0 hardware error to a treu Na
    turbidity = na_if(turbidity, -999.0),
    
    #convert the negative salinity values to true NA
    salinity = if_else(salinity < 0, NA, salinity),  # this will replace any negative salinity values with NA, and any positive values will remain as the unchanged salinity values 
  )

sensor_daily <- 
  sensor_data |>
